@startuml
title Diagrama de Secuencia: Crear pedido de stock

actor Encargado
participant "Interfaz del Sistema" as UI
participant "Controlador de Stock" as Ctrl
participant "Gestor de Stock" as Gestor

' --- Inicio ---
Encargado -> UI : seleccionarOpcion("Crear pedido de stock")
activate UI
UI -> Ctrl : solicitarDatosStock()
activate Ctrl
Ctrl -> Gestor : obtenerExistenciasActuales()
activate Gestor
Gestor --> Ctrl : existenciasActuales
Ctrl -> Gestor : obtenerCantidadRequerida()
Gestor --> Ctrl : cantidadRequerida
deactivate Gestor
Ctrl -> UI : mostrarDiferenciasStock(existenciasActuales, cantidadMensualRequerida)
deactivate Ctrl
deactivate UI

' --- Modificación del pedido ---
Encargado -> UI : modificarCantidades(productosModificados)
activate UI
UI -> Ctrl : validarPedido(productosModificados)
activate Ctrl

alt No hay productos para solicitar
    Ctrl -> UI : mostrarAlerta("No existen productos con faltante de stock")
    deactivate UI
    Ctrl -> UI : solicitarNuevaModificación()
    note left
      Retorna al paso 3 del flujo básico
    end note
else Hay productos con faltante
    Ctrl -> Gestor : generarPedido(productosModificados)
    activate Gestor
    Gestor --> Ctrl : datosPedido(productos, cantidades, costos, proveedores)
    deactivate Gestor

    Ctrl -> UI : mostrarPedidoGenerado(datosPedido)
    deactivate UI
end

deactivate Ctrl

@enduml
