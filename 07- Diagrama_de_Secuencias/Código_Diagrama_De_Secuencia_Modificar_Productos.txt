@startuml
title Diagrama de Secuencia: Actualizar cantidad de productos

actor Encargado
participant "Interfaz del Sistema" as UI
participant "Controlador de Productos" as Ctrl
participant "Gestor de Productos" as Gestor

' --- Inicio ---
Encargado -> UI : seleccionarOpcion("Actualizar cantidad de productos")
activate UI
UI -> Ctrl : solicitarListaProductos()
activate Ctrl
Ctrl -> Gestor : obtenerProductos()
activate Gestor
Gestor --> Ctrl : listaProductos
deactivate Gestor
Ctrl -> UI : mostrarProductos(listaProductos)
deactivate Ctrl
deactivate UI

' --- Actualización ---
Encargado -> UI : modificarCantidad(producto, nuevaCantidad)
activate UI
UI -> Ctrl : validarCantidad(nuevaCantidad)
activate Ctrl

alt Cantidad negativa
    Ctrl -> UI : mostrarAlerta("La cantidad ingresada no es válida")
    deactivate UI
else Cantidad válida
    Ctrl -> Gestor : actualizarCantidad(producto, nuevaCantidad)
    activate Gestor
    deactivate UI
    Gestor --> Ctrl : confirmaciónActualización
    deactivate Gestor

    Ctrl -> UI : mostrarAlerta("Producto actualizado correctamente")


end
deactivate Ctrl
@enduml
