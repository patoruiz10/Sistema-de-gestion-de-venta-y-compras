@startuml
' Versión 2.5 - Diagrama de secuencia: Crear un nuevo usuario

actor Encargado
participant "Interfaz Usuario" as IU
participant "Sistema" as Sistema
participant "Controlador" as Controlador

Encargado -> IU : enviarDatos(nombre, contraseña, tipoUsuario)
activate Sistema
activate IU

IU -> Sistema : enviarDatos(nombre, contraseña, tipoUsuario)
activate Sistema

Sistema -> Sistema : validarPermisos()


alt Permisos válidos
    Sistema -> Controlador : crearNuevoUsuario(datos)
    activate Controlador
    Controlador --> Sistema : confirmación creación
    deactivate Controlador
    
    Sistema -> IU : mostrarMensaje("Usuario creado exitosamente")
    deactivate Sistema
else Permisos inválidos
    Sistema -> IU : mostrarError("Permisos insuficientes")
    deactivate Sistema
end

IU --> Encargado : mostrarResultado()
deactivate IU

@enduml
