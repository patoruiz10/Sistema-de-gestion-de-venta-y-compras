@startuml
!theme plain
title Proceso de Venta - Versión 2.5 (activaciones corregidas para renderizar)

actor "Empleado" as Empleado

participant "frm: FormularioVenta" as frm
participant "cpv: ControladorVenta" as cpv
participant "prod: Producto (Entidad)" as prod
participant "venta: Venta (Entidad)" as venta
participant "pago: MetodoPago (Entidad)" as pago
participant "alerta: VentanaAlerta" as alerta

' -- INICIO --
Empleado -> frm : iniciarProcesoVenta()
activate frm
frm --> Empleado : mostrarFormularioVenta()
' frm activa mientras el usuario ingresa código; se desactiva cuando enviamos al servidor

' =====================================================
' CARGA DE PRODUCTOS
' =====================================================
loop Cargar productos
    Empleado -> frm : ingresarCodigoProducto(codigo)
    ' Desactivar frm antes de pedir al controlador (evita solapamientos)
    deactivate frm

    frm -> cpv : buscarProductoPorID(codigo)
    activate cpv
    cpv -> prod : buscarProductoPorID(codigo)
    activate prod

    alt Producto no encontrado
        prod --> cpv : noEncontrado()
        deactivate prod

        cpv -> alerta : mostrar("Código inexistente.")
        activate alerta
        alerta --> frm : mostrarAlerta("Código inexistente.")
        deactivate alerta

        cpv --> frm : volverAFormulario()
        deactivate cpv

        ' Reactivar frm para que el usuario vuelva a ingresar código
        activate frm
        frm --> Empleado : solicitarNuevoCodigo()
        deactivate frm

    else Producto encontrado
        prod --> cpv : productoEncontrado(datosProducto)
        ' mantengo prod activo hasta devolver respuesta; ahora puedo continuar
        cpv -> venta : agregarProducto(datosProducto)
        activate venta
        venta --> cpv : productoAgregado()
        deactivate venta

        cpv --> frm : actualizarListado()
        deactivate cpv

        ' desactivo prod sólo después de que todo el controlador haya terminado
        deactivate prod

        ' Reactivamos frm para seguir con el siguiente producto
        activate frm
    end
end

' =====================================================
' CÁLCULO TOTAL Y SELECCIÓN DE MÉTODO DE PAGO
' =====================================================
' frm ya está activa aquí (salió del loop)
frm -> cpv : calcularTotal()
deactivate frm
activate cpv

cpv -> venta : obtenerTotal()
activate venta
venta --> cpv : totalCalculado(montoTotal)
deactivate venta

cpv --> frm : mostrarTotal(montoTotal)
deactivate cpv
activate frm

Empleado -> frm : seleccionarMetodoPago(metodo)
frm -> cpv : procesarMetodoPago(metodo)
deactivate frm
activate cpv

cpv -> pago : validarMetodo(metodo)
activate pago

alt Método no disponible
    pago --> cpv : metodoNoDisponible()
    deactivate pago

    cpv -> alerta : mostrar("Método de pago no disponible.")
    activate alerta
    alerta --> frm : mostrarAlerta("Método de pago no disponible.")
    deactivate alerta

    cpv --> frm : volverASeleccionPago()
    deactivate cpv
    activate frm
    frm --> Empleado : seleccionarOtroMetodo()
else Método válido
    pago --> cpv : metodoValido()
    deactivate pago

    ' =====================================================
    ' PROCESO DE PAGO Y FINALIZACIÓN
    ' =====================================================
    cpv -> venta : registrarVenta()
    activate venta

    alt Fondos insuficientes
        venta --> cpv : errorFondos()
        deactivate venta

        cpv -> alerta : mostrar("No es posible efectuar el pago.")
        activate alerta
        alerta --> frm : mostrarAlerta("No es posible efectuar el pago.")
        deactivate alerta

        cpv --> frm : volverASeleccionPago()
        deactivate cpv
        
        frm --> Empleado : seleccionarOtroMetodo()

    else Pago exitoso
        venta --> cpv : ventaRegistrada()
        deactivate venta

        cpv -> alerta : mostrar("Venta concretada correctamente.")
        activate alerta
        alerta --> frm : mostrarAlerta("Venta concretada correctamente.")
        deactivate alerta

        cpv --> frm : limpiarCamposYActualizar()
        deactivate cpv
        
        frm --> Empleado : mostrarTicketYConfirmacion()
        deactivate frm
    end
end

@enduml
