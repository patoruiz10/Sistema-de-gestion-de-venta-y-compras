@startuml
title Diagrama de Secuencia - Proceso de Venta en Minimarket

actor Usuario
participant Sistema
participant Caja
participant Venta
participant LineaDeVenta
participant Producto
participant Inventario
participant ControlStock
participant BaseDeDatos
participant Promocion
participant MetodoDePago

Usuario -> Sistema : solicitarCompra()
activate Sistema

Sistema -> Caja : crearVenta()
activate Caja
Caja -> Venta : nuevaVenta()
activate Venta
Venta --> Caja : ventaCreada
deactivate Venta
deactivate Caja

loop agregar productos
  Sistema -> Producto : seleccionarProducto()
  activate Producto
  Producto -> LineaDeVenta : crearLinea(cantidad, precio)
  activate LineaDeVenta
  LineaDeVenta -> Venta : agregarLinea(l)
  activate Venta
  Venta -> Inventario : descontar(cant)
  activate Inventario
  Inventario --> Venta : ok
  deactivate Inventario
 
  Venta -> ControlStock : verificarNivel(prod)
  activate ControlStock
  ControlStock -> BaseDeDatos : registrarMovimiento(prod, cant)
  activate BaseDeDatos
  BaseDeDatos --> ControlStock : registrado
  deactivate BaseDeDatos
  ControlStock --> Venta : nivelVerificado
  deactivate ControlStock
 
  Venta --> LineaDeVenta : confirmado
  deactivate Venta
  LineaDeVenta --> Producto : lineaCreada
  deactivate LineaDeVenta
  Producto --> Sistema : seleccionado
  deactivate Producto
end

opt aplicar promoción
  Sistema -> Promocion : aplicarAProducto(p)
  activate Promocion
  Promocion -> Venta : actualizarTotal()
  activate Venta
  Venta --> Promocion : totalActualizado
  deactivate Venta
  Promocion --> Sistema : aplicado
  deactivate Promocion
end

Usuario -> MetodoDePago : elegirMetodo()
activate MetodoDePago
MetodoDePago -> MetodoDePago : validar() : boolean
MetodoDePago --> Usuario : validado
deactivate MetodoDePago

Sistema -> Venta : procesarVenta() : boolean
activate Venta
Venta -> Caja : registrarVenta(v) : boolean
activate Caja
Caja --> Venta : registrado
deactivate Caja

Venta -> BaseDeDatos : guardarVenta(v)
activate BaseDeDatos
BaseDeDatos --> Venta : ventaGuardada
deactivate BaseDeDatos

Venta --> Sistema : procesada
deactivate Venta

Sistema -> Caja : cerrar() : boolean
activate Caja
Caja --> Sistema : resultado
deactivate Caja

Sistema -> Usuario : entregarTicket()
deactivate Sistema
@enduml
