@startuml
!theme plain
title Apertura de Caja - Versión 2.6 (Validación unificada)

actor "Empleado" as Empleado

participant "frm: FormularioAperturaCaja" as frm
participant "ccv: ControladorCaja" as ccv
participant "caja: Caja (Entidad)" as caja
participant "alerta: VentanaAlerta" as alerta



' -- Apertura del formulario --
Empleado -> frm : seleccionarOpcion("Apertura de Caja")
activate frm
frm --> Empleado : mostrarFormularioLogin()

' -- Usuario ingresa credenciales --
Empleado -> frm : ingresarCredenciales(usuario, contraseña)

' -- Envío al controlador para verificación --
frm -> ccv : verificarCredenciales(usuario, contraseña)
deactivate frm
activate ccv

' -- Validación en servidor --
ccv -> ccv : validarCredenciales(usuario, contraseña)

alt Credenciales inválidas
    ccv -> alerta : mostrar("Credenciales incorrectas o sin permisos.")
    activate alerta
    alerta --> frm : mostrarAlerta("Credenciales incorrectas.")
    deactivate alerta

    ccv -> frm : volverAFormularioLogin()
    activate frm
    frm --> Empleado : solicitarReingreso()
    deactivate frm
    

else Credenciales válidas
    ' -- Apertura de caja --
    ccv -> caja : abrirCaja(usuario, fechaHoraActual)
    activate caja
    caja --> ccv : aperturaOK
    deactivate caja

    ' -- Confirmación visual --
    ccv -> alerta : mostrar("Caja abierta correctamente (" + fechaHoraActual + ")")
    activate alerta
    alerta --> frm : mostrarAlerta("Caja abierta correctamente.")
    deactivate alerta

    ' -- Actualización de interfaz --
    ccv -> frm : actualizarEstadoCajaAbierta()
    deactivate ccv
    activate frm
    deactivate ccv
    frm --> Empleado : mostrarConfirmacion()
    deactivate frm
end


@enduml
