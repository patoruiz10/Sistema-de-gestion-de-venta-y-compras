@startuml
title Diagrama de Secuencia: Consulta de historial de ventas 

actor Encargado
participant "Interfaz del Sistema" as UI
participant "Controlador de Ventas" as Ctrl
participant "Gestor de Ventas" as Gestor

' --- Inicio ---
Encargado -> UI : seleccionarOpcion("Consulta de historial de ventas")
activate UI
UI -> Ctrl : solicitarOpcionesConsulta()
activate Ctrl
Ctrl -> UI : mostrarOpcionesConsulta([día, semana, mes])
deactivate Ctrl
deactivate UI

' --- Validación ---
Encargado -> UI : ingresarTipoConsulta(opcion, fecha)
activate UI
UI -> Ctrl : validarTipoConsulta(opcion, fecha)
activate Ctrl
deactivate UI

' --- Validaciones de fecha ---
alt Fecha futura
    Ctrl -> UI : mostrarAlerta("La fecha ingresada es futura")
    activate UI
    UI -> Encargado : mostrarMensaje("La fecha ingresada es futura")
    deactivate UI

else Fecha inválida o incompleta
    Ctrl -> UI : mostrarAlerta("Debe ingresar una fecha válida")
    activate UI
    UI -> Encargado : mostrarMensaje("Debe ingresar una fecha válida")
    deactivate UI

else Fecha válida
    Ctrl -> Gestor : consultarVentas(opcion, fecha)
    activate Gestor
    Gestor --> Ctrl : listaVentas
    deactivate Gestor

    ' --- Validación de resultados ---
    alt listaVentas vacía
        Ctrl -> UI : mostrarAlerta("No se encontraron ventas registradas")
        activate UI
        UI -> Encargado : mostrarMensaje("No se encontraron ventas registradas")
        deactivate UI

    else listaVentas con datos
        Ctrl -> UI : mostrarVentas(listaVentas)
        activate UI
        UI -> Encargado : mostrarHistorial(listaVentas)
        deactivate UI
    end
end

deactivate Ctrl

@enduml
