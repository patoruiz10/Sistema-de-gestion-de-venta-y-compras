' ==== CLASES ====

class Producto {
  - nombre : string
  - costo : float
  - codigoBarra : int
  - categoria : string
  - cantidad : int
  - precioUnitario : float
  + cambiarPrecio(nuevo : float)
  + asignarCategoria()
}

class Categoría {
    -nombre
    - productos : List<Producto>
    + eliminarProducto(p : Producto)
    + agregarProducto (p : Producto)
}

class Cliente {
 - condicionIVA : string
 - nombre : String
 - telefono : int
 - dni : int
 + mostrarInfo()
 + compra()
}

class Venta {
  - fecha : Date
  - total : float
  - lineaVenta : String
  - descuento
  - pago
  + agregarMetododePago(MetodoDePago)
  + agregarDatos(Cliente)
  + agregarLineaVenta(LineaDeVenta)
  + eliminarLinea(LineaDeVenta)
  + registraPagos(mdp : int)
  + obtenerSubtotal() : float
  + aplicarDescuento(d : float)
  + quitarDescuento(d : float)
  + calcularTotal(Subtotal , condicionIVA, descuento : float) : float
}

class LineaDeVenta {
  - cantidad : int
  - precioUnitario : float
  - Subtotal : float
  + actualizarCantidad(cant : int)
  + actualizarPrecioUnitario (precioU : float)
  + calcularSubtotal() : float
}

class Empleado {
  - nombre : String
  - telefono : int
  - dni : int
  - rol : String
  + mostrarInfo()
  + iniciaVenta()
  + finalizaVenta() : boolean 
}

class Caja {
  - numero : int
  - monto: float
  - estado: bool
  + balance()
  + abrir() : boolean
  + cerrar() : boolean
}

class MetodoDePago {
  - tipo : int
  - detalles : String
  + obtenerTipo()
}
class Impresion {
    - nroTicket : int
    - empleado : Empleado
    + generarNroTicket () : int 
    + imprimirTicket()
}

class Controlador {
   + inicioSesion(string) : boolean
   + cierraSesion() : boolean
   + validarPago() : boolean
   + procesarPago()
   + añadirProducto(p : Producto)
   + eliminarProducto(p : Producto)
   + buscarProducto(p : Producto)
   + actualizarProducto(p : Producto)
   + guardarComprobante(v : Venta)
   + solicitarReabastecimiento()
}
class Persona{
  - nombre : String
  - telefono : int
  - dni : int
  + mostrarInfo()
}
class Proveedor {
  - mail : string
  - cuit : int
  - telefono : int
  - direccion : String
  
  + mostrarInfo() : String
  + contactar() : void
  + registrarEntrega(fecha : Date) : void
  
}
