' ==== CLASES ====

class Producto {
  - id : UUID
  - nombre : String
  - precio : Money
  - codigoBarra : String
  + cambiarPrecio(nuevo : Money)
}

class Categoria {
  - id : UUID
  - nombre : String
  + agregarProducto(p : Producto)
  + quitarProducto(p : Producto)
}

class Especificacion {
  - clave : String
  - valor : String
  + actualizarValor(v : String)
}

class Inventario {
  - id : UUID
  - cantidad : int
  + descontar(qty : int)
  + reponer(qty : int)
  + verificarStock() : boolean
}

class Cliente {
  - id : UUID
  - nombre : String
  - email : String
  + realizarVenta(v : Venta)
  + devolverVenta(d : Devolucion)
}

class Venta {
  - id : UUID
  - fecha : Date
  - total : Money
  + calcularTotal() : Money
  + agregarLinea(l : LineaDeVenta)
  + eliminarLinea(l : LineaDeVenta)
  + aplicarMetodoPago(m : MetodoDePago)
  + procesarVenta()
}

class LineaDeVenta {
  - cantidad : int
  - precioUnitario : Money
  + calcularSubtotal() : Money
  + actualizarCantidad(qty : int)
}

class Empleado {
  - id : UUID
  - nombre : String
  - rol : String
  + procesarVenta(v : Venta)
  + registrarDevolucion(d : Devolucion)
}

class Caja {
  - id : UUID
  - numero : int
  + registrarVenta(v : Venta)
  + abrir()
  + cerrar()
}

class Proveedor {
  - id : UUID
  - nombre : String
  - telefono : String
  + enviarProducto(p : Producto, qty : int)
  + recibirPago(amount : Money)
}

class Promocion {
  - id : UUID
  - descripcion : String
  - descuento : double
  + aplicarAProducto(p : Producto)
  + quitarDeProducto(p : Producto)
}

class Devolucion {
  - id : UUID
  - fecha : Date
  - motivo : String
  + procesar()
}

class MetodoDePago {
  - tipo : String
  - detalles : String
  + validar() : boolean
  + registrarPago(amount : Money)
}
