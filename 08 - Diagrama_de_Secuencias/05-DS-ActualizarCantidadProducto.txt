@startuml
!theme plain
title Actualizar Cantidad de Producto - SELVENT

actor Encargado

participant Controlador
participant Producto

'===========================================================
' INICIO DEL PROCESO
'===========================================================

Encargado -> Controlador : seleccionarOpcionActualizarCantidad()
activate Controlador

Controlador --> Encargado : <<"Mostrar listado de productos para actualizar":string>>
deactivate Controlador
'===========================================================
' INGRESO DE DATOS DE ACTUALIZACIÓN
'===========================================================

Encargado -> Controlador : modificarCantidad(codigo, nuevaCantidad)
activate Controlador
'===========================================================
' VALIDACIÓN DE CANTIDAD
'===========================================================

alt Cantidad negativa o no válida
    Controlador --> Encargado : <<"La cantidad ingresada no es válida":string>>

else Cantidad válida

    '=======================================================
    ' BÚSQUEDA DEL PRODUCTO POR CÓDIGO
    '=======================================================
    Controlador -> Producto : obtenerCodigo(codigo)
    activate Producto

    alt Producto no encontrado
        Producto --> Controlador : <<"Producto no encontrado":string>>
        

        Controlador --> Encargado : <<"El código de producto ingresado no existe":string>>

    else Producto encontrado
        Producto --> Controlador : <<producto>>
        deactivate Producto

        '===================================================
        ' ACTUALIZACIÓN DE LA CANTIDAD EN STOCK
        '===================================================
        Controlador -> Producto : actualizarCantidadStock(nuevaCantidad : int)
        activate Producto
        Producto --> Controlador : <<"Cantidad actualizada correctamente":string>>
        deactivate Producto

        Controlador --> Encargado : <<"Producto actualizado correctamente":string>>
        deactivate Controlador
    end
end



@enduml
