@startuml
!theme plain
title Diagrama de Secuencia - Consultar Variación del Stock

actor "e:Empleado" as Empleado
participant "c:Controlador" as Controlador
participant "<<multiobjeto>> cat:Categorias" as Categorias
participant "hsd:HistorialStockDiario" as Historial

'===========================================================
' INICIO DEL PROCESO (CONSULTA GENERAL)
'===========================================================
Empleado -> Controlador : consultarVariacionStock(fecha)
activate Controlador

'===========================================================
' OBTENER CATÁLOGO COMPLETO
'===========================================================
Controlador -> Categorias : obtenerTodosLosProductos()
activate Categorias
Categorias --> Controlador : listaProductos : List<Producto>
deactivate Categorias

'===========================================================
' LOOP: POR CADA PRODUCTO, OBTENER DATOS ACTUAL Y MES ANTERIOR
'===========================================================
loop "Por cada producto en listaProductos"
  Controlador -> Historial : obtenerStockEnFecha(codigo, fecha)
  activate Historial
    Historial --> Controlador : <<cantidadEnFecha:int>>
  deactivate Historial

  note right of Controlador
    variacion =  cantidadEnFecha - cantidadActual
    listaVariaciones = codigo, descripcion, cantidadActual, cantidadEnFecha, variacion
  end note
end

Controlador --> Empleado : <<listaVariaciones, "Variación de stock mostrada correctamente":string>>
deactivate Controlador


@enduml
