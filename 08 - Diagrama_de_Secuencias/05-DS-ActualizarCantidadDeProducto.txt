@startuml
!theme plain
title Diagrama de Secuencia - Actualizar Cantidad de Producto

actor "e:Empleado" as Empleado
participant "c:Controlador" as Controlador
participant "<<multiobjeto>> categorias:Categoria" as Categorias
participant "prod:Producto" as Producto

'===========================================================
' INGRESO DE DATOS DE ACTUALIZACIÓN
'===========================================================
Empleado -> Controlador : modificarCantidad(codigo, nuevaCantidad)
activate Controlador

'===========================================================
' VALIDACIÓN DE CANTIDAD
'===========================================================
alt nuevaCantidad < 0
  Controlador --> Empleado : <<"La cantidad ingresada no es válida.":string>>
else nuevaCantidad >= 0

  '=======================================================
  ' BÚSQUEDA DEL PRODUCTO POR CÓDIGO (LOOP)
  '=======================================================
  loop "Buscar producto por código"
    Controlador -> Categorias : buscarCodigo(codigo)
    activate Categorias
    Categorias --> Controlador : <<idProducto:int>>
    deactivate Categorias

    note right of Controlador
      buscarCodigo(codigo):
      - recorre internamente categorías y sus productos
      - si encuentra coincidencia, devuelve <<idProducto>>
      - si no encuentra, devuelve <<null>>
      - corta iteración al encontrar un id válido
    end note
  end

  alt idProducto == null
    Controlador --> Empleado : <<"El código de producto ingresado no existe":string>>
  else idProducto != null

    Controlador -> Producto : modificarCantidadStock(nuevaCantidad)
    activate Producto
    Producto --> Controlador : <<"Cantidad actualizada correctamente":string>>
    deactivate Producto

    Controlador --> Empleado : <<"Producto actualizado correctamente":string>>
    deactivate Controlador
  end
end

@enduml
