@startuml
!theme plain
title Diagrama de Secuencia - Mostrar Inventario

actor "e:Empleado" as Empleado
participant "c:Controlador" as Controlador
participant "<<multiobjeto>> categorias:Categorias" as Categorias

'===========================================================
' INICIO DEL PROCESO
'===========================================================
Empleado -> Controlador : mostrarInventario()
activate Controlador

'===========================================================
' OBTENER INVENTARIO COMPLETO
'===========================================================
Controlador -> Categorias : obtenerTodosLosProductos()
activate Categorias
Categorias --> Controlador : << inventarioCompleto : List<Producto> >>
deactivate Categorias

'===========================================================
' MOSTRAR INVENTARIO AL USUARIO
'===========================================================
Controlador --> Empleado : <<inventarioCompleto: List<Producto>, "Inventario mostrado correctamente":string>>
deactivate Controlador
@enduml
