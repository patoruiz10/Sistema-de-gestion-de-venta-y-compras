@startuml
!theme plain
title Diagrama de Secuencia - Consulta de Historial de Ventas 

actor "e:Empleado" as Empleado
participant "c:Controlador" as Controlador
participant "<<multiobjeto>> cs:Caja" as Cajas
participant "cj:Caja" as Caja

'===========================================================
' USUARIO INGRESA FECHA
'===========================================================
Empleado -> Controlador : consultarHistorialVentas(fecha)
activate Controlador

'===========================================================
' VALIDACIONES
'===========================================================
alt Fecha == null
  Controlador --> Empleado : <<"Falta completar la fecha":string>>


else Fecha futura al dia actual
  Controlador --> Empleado : <<"La fecha ingresada es futura":string>>


else Fecha válida
  '=========================================================
  ' CONSULTAR VENTAS POR FECHA (TODAS LAS CAJAS)
  '=========================================================
  Controlador -> Cajas : obtenerVentasPorFecha(fecha)
  activate Cajas

  loop "Por cada caja del sistema"
    Cajas -> Caja : obtenerVentasPorFecha(fecha)
    activate Caja
    Caja --> Cajas : ventasCaja : List<Venta>
    deactivate Caja
  end
  
note right of Cajas
  Se acumulan las listas
  ventasCaja en listaVentas
end note

  Cajas --> Controlador : listaVentas : List<Venta>
  deactivate Cajas

Controlador --> Empleado : <<listaVentas, "Ventas mostradas correctamente":string>>

  deactivate Controlador
end

@enduml
