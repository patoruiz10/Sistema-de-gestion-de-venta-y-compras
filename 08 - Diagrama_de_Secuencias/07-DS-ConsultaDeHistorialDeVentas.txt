@startuml
!theme plain
title Diagrama de Secuencia - Consulta de Historial de Ventas

actor "e:Empleado" as Empleado
participant "c:Controlador" as Controlador
participant "v:Venta" as Venta

'===========================================================
' USUARIO INGRESA FECHA
'===========================================================
Empleado -> Controlador : consultarHistorialVentas(fecha)
activate Controlador

'===========================================================
' VALIDACIONES
'===========================================================
alt Fecha futura al dia actual
  Controlador --> Empleado : <<"La fecha ingresada es futura":string>>
else Fecha válida
  '=========================================================
  ' CONSULTAR VENTAS POR FECHA
  '=========================================================
  Controlador -> Venta : obtenerVentasPorFecha(fecha)
  activate Venta
  Venta --> Controlador : <<listaVentas:List<Venta>>>
  deactivate Venta

  alt listaVentas vacía
    Controlador --> Empleado : <<"No se encontraron ventas registradas en la fecha consultada":string>>
  else listaVentas con datos
    Controlador --> Empleado : <<listaVentas:List<Venta>>>, <<"Ventas mostradas correctamente":string>>
    deactivate Controlador
  end
end

@enduml
