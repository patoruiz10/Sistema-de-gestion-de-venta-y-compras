participant Controlador

Empleado -> Controlador : seleccionarOpcionCrearPedidoStock()

'===========================================================

'===========================================================

Controlador -> Producto : obtenerCantidad()

Controlador -> Producto : obtenerCantidadRequerida()
activate Producto
Producto --> Controlador : <<cantidadRequerida>>
deactivate Producto
Controlador --> Empleado : <<null>>
' Controlador calcula faltantes
'===========================================================
' MODIFICACIÓN DEL PEDIDO POR EL USUARIO
'===========================================================
'===========================================================
' VALIDACIÓN
'===========================================================
opt No hay productos con faltante
    Controlador --> Empleado : <<"No existen productos con faltante de stock":string>>
    Controlador --> Empleado : <<"Ingrese nuevas cantidades":string>>
    deactivate Controlador    
end

    Empleado -> Controlador : modificarCantidades(productosModificados)
    activate Controlador
    
    '=======================================================
    ' GENERAR PEDIDO
    '=======================================================
    Controlador -> Proveedor : obtenerProveedor(cuit)
    activate Proveedor
    Proveedor --> Controlador : <<proveedor>>
    deactivate Proveedor
