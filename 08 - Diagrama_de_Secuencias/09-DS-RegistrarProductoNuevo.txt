@startuml
!theme plain
title Diagrama de Secuencia - Registrar Producto Nuevo

actor "e:Empleado" as Empleado
participant "c:Controlador" as Controlador
participant "<<multiobjeto>> cat:Categoria" as Categorias
participant "<<multiobjeto>> prov:Proveedor" as Proveedores
participant "<<multiobjeto>> prod:Producto" as Productos
participant "cat:Categoria" as cat

'===========================================================
' USUARIO COMPLETA FORMULARIO Y PRESIONA GUARDAR
'===========================================================
Empleado -> Controlador : añadirProducto(codigo, descripcion, cantidad, precioUnitario, cuit, nombreCategoria)
activate Controlador

'===========================================================
' VALIDACIÓN UNIFICADA DE CAMPOS VACÍOS
'===========================================================
alt Algún campo obligatorio está vacío
  Controlador --> Empleado : <<"Faltan completar campos obligatorios":string>>
else Todos los campos completos

  '===========================================================
  ' 1) VALIDACIÓN DEL CÓDIGO (NO DEBE EXISTIR)
  '===========================================================
  loop "Para cada producto en Categorias"
    Controlador -> Categorias : buscarCodigo(codigo)
    activate Categorias
    Categorias --> Controlador : <<idProducto:int>>
    deactivate Categorias

    note right of Controlador
      buscarCodigo(codigo):
      - recorre categorías y sus productos
      - si encuentra coincidencia, devuelve <<idProducto>>
      - si no encuentra, devuelve <<null>>
      - corta iteración al encontrar un id distinto de null.
    end note
  end

  alt idProducto != null
    Controlador --> Empleado : <<"El codigo del producto ingresado ya existe":string>>
  else idProducto == null

    '===========================================================
    ' 2) VALIDACIÓN DEL PROVEEDOR (DEBE EXISTIR)
    '===========================================================
    loop "Para cada proveedor en Proveedores"
      Controlador -> Proveedores : obtenerProveedor(cuit)
      activate Proveedores
      Proveedores --> Controlador : <<cuitProv:int>>
      deactivate Proveedores

      note right of Controlador
        obtenerProveedor(cuit):
        - recorre proveedores
        - si encuentra coincidencia, devuelve <<cuitProv>>
        - si no encuentra, devuelve <<null>>
        - corta iteración al encontrar un cuit valido.
      end note
    end

    alt cuitProv == null
      Controlador --> Empleado : <<"El proveedor seleccionado no existe":string>>
    else cuitProv != null

      '===========================================================
      ' 3) VALIDACIÓN DE LA CATEGORÍA (DEBE EXISTIR)
      '===========================================================
      loop "Para cada categoría en Categorias"
        Controlador -> Categorias : obtenerCategoria(nombreCategoria)
        activate Categorias
        Categorias --> Controlador : <<cat:Categoria>>
        deactivate Categorias

        note right of Controlador
          obtenerCategoria(nombreCategoria):
          - recorre categorias
          - si encuentra coincidencia, devuelve <<cat:Categoria>>
          - si no encuentra, devuelve <<null>>
          - corta iteración al encontrar una categoria valida.
        end note
      end

      alt cat == null
        Controlador --> Empleado : <<"La categoría seleccionada no existe":string>>
      else cat != null

        '===========================================================
        ' 4) CREAR OBJETO PRODUCTO (NUEVA ENTIDAD)
        '===========================================================
        create "productoNuevo : Producto" as productoNuevo
        Controlador -> productoNuevo : create(codigo, descripcion, cantidad, precioUnitario, cuit, nombreCategoria)

        '===========================================================
        ' 5) AGREGAR PRODUCTO AL MULTIOBJETO DE PRODUCTOS
        '===========================================================
        Controlador -> Productos : agregarProducto(productoNuevo : Producto)
        activate Productos
        Productos --> Controlador : <<null>>
        deactivate Productos

        '===========================================================
        ' 6) AGREGAR PRODUCTO A LA CATEGORÍA SELECCIONADA
        '===========================================================
        Controlador -> cat : agregarProducto(productoNuevo : Producto)
        activate cat
        cat --> Controlador : <<null>>
        deactivate cat

        Controlador --> Empleado : <<"Producto registrado correctamente":string>>
        deactivate Controlador
      end
    end
  end
end

@enduml
