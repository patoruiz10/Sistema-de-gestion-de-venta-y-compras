@startuml
!theme plain
title Registrar Proveedor Nuevo - Versión 2.5 (activaciones corregidas)

actor "Empleado" as Empleado

participant "frm: FormularioProveedor" as frm
participant "cpp: ControladorProveedor" as cpp
participant "prov: Proveedor (Entidad)" as prov
participant "alerta: VentanaAlerta" as alerta


' ====== INICIO ======
Empleado -> frm : abrirFormularioRegistrarProveedor()
activate frm
frm --> Empleado : mostrarFormulario()

' ====== INGRESO DE DATOS ======
Empleado -> frm : ingresarDatosProveedor(codigo, compañia, contacto, teléfono, detalles)

' Validación de campos en el formulario
frm -> frm : validarCampos()

alt Falta código de proveedor
    frm -> alerta : mostrar("Debe ingresar el código del proveedor.")
    activate alerta
    alerta --> frm : cerrarAlerta()
    deactivate alerta
    frm --> Empleado : solicitarCodigoFaltante()
else Datos completos
    ' ====== ENVÍO AL CONTROLADOR ======
    frm -> cpp : registrarProveedor(datosProveedor)
    deactivate frm
    activate cpp

    ' ====== VALIDACIÓN DE CÓDIGO ======
    cpp -> prov : buscarPorCodigo(datosProveedor.codigo)
    activate prov

    alt Código repetido
        prov --> cpp : proveedorExistente()
        deactivate prov

        cpp -> alerta : mostrar("El código ingresado ya existe.")
        activate alerta
        alerta --> frm : mostrarAlerta("El código ingresado ya existe.")
        deactivate alerta

        cpp --> frm : volverAFormulario()
        deactivate cpp

        activate frm
        frm --> Empleado : corregirCodigo()
        deactivate frm

    else Código disponible
        prov --> cpp : noEncontrado()
        deactivate prov

        ' ====== REGISTRO DEL NUEVO PROVEEDOR ======
        cpp -> prov : guardar(datosProveedor)
        activate prov
        prov --> cpp : proveedorGuardado()
        deactivate prov

        ' ====== CONFIRMACIÓN ======
        cpp -> alerta : mostrar("Proveedor registrado correctamente.")
        activate alerta
        alerta --> frm : mostrarAlerta("Proveedor registrado correctamente.")
        deactivate alerta

        cpp --> frm : actualizar()
        deactivate cpp

        activate frm
        frm --> Empleado : confirmarRegistroExitoso()
        deactivate frm
    end
end

@enduml
