@startuml
!theme plain
title  Diagrama de Secuencia - Proceso de Venta

actor "e:Empleado" as Empleado
participant "c:Controlador" as Controlador
participant "cj:Caja" as Caja
participant "v:Venta" as Venta
participant "prod:Producto" as Producto
participant "mdp:MetodoDePago" as MetodoDePago
participant "i:Impresion" as Impresion
participant "<<multiobjeto>> categorias:Categoria" as categorias
participant "<<multiobjeto>> LineasDeVenta : LineaDeVenta" as LineasDeVenta

'===========================================================
' INICIO DEL PROCESO DE VENTA
'===========================================================
Empleado -> Controlador : iniciarVenta()
activate Controlador

Controlador -> Venta : agregarDatos(Cliente)
activate Venta
Venta --> Controlador : <<"Venta iniciada":string>>
deactivate Venta

'===========================================================
' CARGA DE PRODUCTOS
'===========================================================
loop "Cargar productos mientras el usuario desee agregar"

    Empleado -> Controlador : ingresarCodigoProducto(codigo)

    '-------------------------------------------------------
    ' BÚSQUEDA DEL ID DEL PRODUCTO
    '-------------------------------------------------------
    loop "para cada producto en categorias."
    Controlador -> categorias : buscarCodigo(codigo)
        activate categorias
    
    
            categorias --> Controlador : <<idProducto>>
            deactivate categorias
            note left of categorias
              buscarCodigo(codigo):
              - recorre internamente categorías y sus productos
              - si encuentra coincidencia, devuelve <<idProducto>>
              - si no encuentra, devuelve <<null>>
              - corta iteración al encontrar un id valido.
            end note
        
    end

    

    alt idProducto == null
        Controlador --> Empleado : <<"Código inexistente, vuelva a ingresar":string>>

    else idProducto != null
        '-------------------------------------------------------
        ' OBTENER PRODUCTO COMPLETO EN OTRO LADO (Producto)
        '-------------------------------------------------------
        Controlador -> Producto : obtenerProductoPorId(idProducto)
        activate Producto
        Producto --> Controlador : <<prod:Producto>>
        deactivate Producto

        Controlador --> Empleado : <<prod:Producto>>
        deactivate Controlador
        '-------------------------------------------------------
        ' OBTENER DATOS DEL PRODUCTO
        '-------------------------------------------------------

        ' El usuario ingresa cantidad
        Empleado -> Controlador : ingresarCantidad(cant)
        activate Controlador
        '-------------------------------------------------------
        ' CREAR LÍNEA DE VENTA
        '-------------------------------------------------------
        create "lineaNueva : LineaDeVenta" as lineaNueva
        Controlador -> lineaNueva : crearLineaVenta(prod:Producto, cant)

        ' Registrar línea en el multiobjeto de líneas
        Controlador -> LineasDeVenta : agregarLineaVenta(lineaNueva)
        activate LineasDeVenta
        LineasDeVenta --> Controlador : <<null>>
        deactivate LineasDeVenta

        ' Agregar línea a la venta
        Controlador -> Venta : agregarLineaVenta(lineaNueva)
        activate Venta
        Venta --> Controlador : <<"Línea de venta agregada":string>>
        deactivate Venta

        '===========================================================
        ' CÁLCULO DEL TOTAL
        '===========================================================
        Controlador -> Venta : calcularTotal()
        activate Venta
        Venta --> Controlador : <<total>>
        deactivate Venta
        Controlador --> Empleado : <<"Producto agregado a la venta":string , total>>
        deactivate Controlador
    end

end



'===========================================================
' SELECCIÓN Y VALIDACIÓN DEL MÉTODO DE PAGO
'===========================================================
Empleado -> Controlador : seleccionarMetodoPago(tipo)
activate Controlador



loop "para cada tipo disponible."
    Controlador -> MetodoDePago : obtenerTipo(tipo)
    activate MetodoDePago
    
    MetodoDePago --> Controlador : <<tipoDisponible : MetodoDePago>>
        deactivate MetodoDePago
    
    note left of MetodoDePago
      obtenerTipo():
      - recorre los metodos de pagos disponibles
      - si tipoDisponible == tipo => valido
      - si no => null
      - corta iteración al ser valido el tipo
    end note
end

alt tipoDisponible == null

    Controlador --> Empleado : <<"Método de pago no disponible":string>>

else tipoDisponible != null
    deactivate MetodoDePago

    Controlador -> Venta : registraPago(tipoDisponible)
    activate Venta

    alt Fondos insuficientes
        Venta --> Controlador : <<"No es posible efectuar el pago":string>>
        Controlador --> Empleado : <<"No es posible efectuar el pago":string>>
        

    else Proceso de pago cancelado
        Venta --> Controlador : <<"Proceso de pago cancelado":string>>
        Controlador --> Empleado : <<"Proceso de pago cancelado":string>>
        

    else Pago aceptado
        Venta --> Controlador : <<"Pago registrado":string>>
        deactivate Venta

        '===================================================
        ' POST-PAGO: por cada línea -> actualizar stock + balance()
        '===================================================
        loop "Para cada línea de venta de la venta actual"

            note left of Controlador
              Se descuenta la cantidadVendida de cada producto vendido.
            end note

            Controlador -> Producto : actualizarCantidadStock(idProducto, cantidadVendida)
            activate Producto
            Producto --> Controlador : <<null>>
            deactivate Producto

            Controlador -> Caja : balance()
            activate Caja
            Caja --> Controlador : <<saldoActual>>
            deactivate Caja
        end

        '===================================================
        ' Comprobante digital
        '===================================================
        Controlador -> Impresion : generarNroTicket()
        activate Impresion
        Impresion --> Controlador : <<nroTicket>>
        deactivate Impresion

        Controlador -> Impresion : guardarComprobante(v:Venta, nroTicket)
        activate Impresion
        Impresion --> Controlador : <<"Comprobante guardado":string>>
        deactivate Impresion

        Controlador --> Empleado : <<"Venta registrada correctamente":string>>
        deactivate Controlador
    end
end

@enduml
