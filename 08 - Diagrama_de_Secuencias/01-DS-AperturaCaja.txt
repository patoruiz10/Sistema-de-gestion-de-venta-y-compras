@startuml
!theme plain
title Diagrama de Secuencia - Apertura de Caja

actor "e:Empleado" as Empleado
participant "c:Controlador" as Controlador
participant "cj:Caja" as Caja

' --- Inicio del caso ---

' --- Validación de credenciales ---
Empleado -> Controlador : inicioSesion(usuario, contraseña)
activate Controlador

alt Credenciales inválidas
    Controlador --> Empleado : <<"Credenciales incorrectas" :string>>
else Credenciales válidas



' --- Verificación del estado de la caja ---
Controlador -> Caja : abrirCaja()
activate Caja
Caja --> Controlador : <<estado :bool>>
deactivate Caja


    ' --- Registrar apertura ---

    Controlador --> Empleado : <<"Caja abierta correctamente" :string>>
    deactivate Caja
    deactivate Controlador
end
    

@enduml
