@startuml
!theme plain
title Diagrama de Secuencia - Crear Pedido de Stock

actor "e:Empleado" as Empleado
participant "c:Controlador" as Controlador
participant "prod:Producto" as Producto
participant "<<multiobjeto>> categorias:Categoria" as categorias

'===========================================================
' INICIO DEL PROCESO
'===========================================================

Empleado -> Controlador : crearPedidoStock()
activate Controlador



'===========================================================
' OBTENER STOCK ACTUAL Y REQUERIDO
'===========================================================
loop "para cada producto de cantidad <= 10"
    Controlador -> categorias : obtenerProducto()
    activate categorias
    categorias --> Controlador : <<prod:producto>>
    deactivate categorias

    Controlador --> Empleado : <<prod:producto>>
end
' Controlador calcula faltantes
'===========================================================
' MODIFICACIÓN DEL PEDIDO POR EL USUARIO
'===========================================================
'===========================================================
' VALIDACIÓN
'===========================================================
opt No hay productos con cantidad <= 10
    Controlador --> Empleado : <<"No existen productos con faltante de stock":string>>
    deactivate Controlador    
end

Empleado -> Controlador : modificarCantidades(productosModificados)
activate Controlador

'=======================================================
' ARMADO DEL PEDIDO (producto + proveedor + cantidades)
'=======================================================
loop "para cada prod en productosModificados"
note right of Empleado
    cantidadAReponer = cantidad fijada por el empleado
    
end note
    Controlador -> Producto : obtenerProveedor()
    activate Producto
    Producto --> Controlador : <<prov:proveedor>>
    deactivate Producto

    Controlador --> Empleado : <<prod:Producto>>, <<cantidadAReponer>>, <<prov:proveedor>>
    
deactivate Controlador
end

@enduml
