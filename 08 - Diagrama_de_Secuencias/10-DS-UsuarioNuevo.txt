@startuml
!theme plain
title Diagrama de Secuencia - Crear nuevo usuario

actor "e:Empleado" as Empleado
participant "c:Controlador" as Controlador
participant "<<multiobjeto>> em : Empleado" as Usuarios

'===========================================================
' INGRESO DE DATOS
'===========================================================
Empleado -> Controlador : crearNuevoUsuario(nombre, telefono, dni, usuario, contraseña, rol)
activate Controlador

'===========================================================
' VALIDACIÓN DE CAMPOS VACÍOS
'===========================================================
alt Campos vacíos
    Controlador --> Empleado : <<"Faltan completar campos obligatorios":string>>

else Campos válidos

    '=======================================================
    ' CREACIÓN DEL NUEVO EMPLEADO (NUEVA ENTIDAD)
    '=======================================================
    create "nuevoUsuario : Empleado" as nuevoEmpleado
    Controlador -> nuevoEmpleado : create(nombre, telefono, dni, usuario, contraseña, rol)
    
    '=======================================================
    ' AGREGAR AL MULTIOBJETO DE USUARIOS
    '=======================================================
    Controlador -> Usuarios : agregarUsuario(nuevoEmpleado: Empleado)
    activate Usuarios
    Usuarios --> Controlador : <<null>>
    deactivate Usuarios

    '=======================================================
    ' CONFIRMACIÓN DE REGISTRO         
    '=======================================================
    Controlador --> Empleado : <<"Usuario creado correctamente":string>>
    deactivate Controlador

end

@enduml
