@startuml
!theme plain
title Diagrama de Secuencia - Crear nuevo usuario

actor Empleado
participant Controlador
participant "<<multiobjeto>> Usuarios : Empleado" as Usuarios

'===========================================================
' INICIO DEL PROCESO
'===========================================================
Empleado -> Controlador : seleccionarOpcionCrearNuevoUsuario()
activate Controlador
Controlador --> Empleado : <<null>>
deactivate Controlador

'===========================================================
' INGRESO DE DATOS
'===========================================================
Empleado -> Controlador : crearNuevoUsuario(usuario, contraseña, rol)
activate Controlador

'===========================================================
' VALIDACIÓN DE CAMPOS VACÍOS
'===========================================================
alt Campos vacíos
    Controlador --> Empleado : <<"Faltan completar campos obligatorios":string>>

else Campos válidos

    '=======================================================
    ' CREACIÓN DEL NUEVO EMPLEADO (NUEVA ENTIDAD)
    '=======================================================
    create "nuevoUsuario : Empleado" as nuevoEmpleado
    Controlador -> nuevoEmpleado : crearUsuario(usuario, contraseña, rol)
    

    '=======================================================
    ' AGREGAR AL MULTIOBJETO DE USUARIOS
    '=======================================================
    Controlador -> Usuarios : agregarUsuario(nuevoUsuario: Empleado)
    activate Usuarios
    Usuarios --> Controlador : <<null>>
    deactivate Usuarios

    '=======================================================
    ' CONFIRMACIÓN DE REGISTRO
    '=======================================================
    Controlador --> Empleado : <<"Usuario creado correctamente":string>>
    deactivate Controlador

end

@enduml
