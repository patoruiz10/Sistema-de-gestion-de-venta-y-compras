@startuml
title Proceso de Venta

actor Empleado

participant "frm: FormularioVenta" as frm
participant "cpv: ControladorVenta" as cpv
participant "prod: Producto (Entidad)" as prod
participant "venta: Venta (Entidad)" as venta
participant "pago: MetodoPago (Entidad)" as pago
participant "alerta: VentanaAlerta" as alerta

Empleado -> frm : iniciaProcesoVenta()
activate frm
frm -> Empleado : mostrarFormularioVenta()

loop Cargar productos
    Empleado -> frm : ingresarCodigoProducto(codigo)
    frm -> cpv : buscarProductoPorID(codigo)
    activate cpv
    cpv -> prod : buscarProductoPorID(codigo)
    activate prod
    alt Producto no encontrado
        prod --> cpv : noEncontrado()
        deactivate prod
        cpv -> alerta : mostrarAlerta("Código inexistente.")
        activate alerta
        alerta --> frm : mostrar("Código inexistente.")
        deactivate alerta
        frm -> Empleado : solicitarNuevoCodigo()
    else Producto encontrado
        prod --> cpv : productoEncontrado(datosProducto)
        deactivate prod
        cpv -> frm : agregarProducto(datosProducto)
        deactivate cpv
        frm -> frm : actualizarListado()
    end

end

frm -> cpv : calcularTotal()
activate cpv
cpv -> venta : obtenerTotal()
activate venta
venta --> cpv : totalCalculado(montoTotal)
deactivate venta
cpv -> frm : mostrarTotal(montoTotal)
deactivate cpv

frm -> cpv : seleccionarMetodoPago(metodo)
activate cpv
cpv -> pago : procesarMetodoPago(metodo)
activate pago
pago -> pago : validarMetodo(metodo)
alt Método no disponible
    pago --> cpv : metodoNoDisponible()
    deactivate pago
    cpv -> alerta : mostrarAlerta("Método de pago no disponible.")
    activate alerta
    alerta --> frm : mostrar("Método de pago no disponible.")
    deactivate alerta
    frm -> Empleado : seleccionarOtroMetodo()
else Método válido
    pago --> cpv : metodoValido()
    deactivate pago
    cpv -> venta : registrarVenta()
    activate venta
    alt Fondos insuficientes
        venta --> cpv : errorFondos()
        deactivate venta
        cpv -> alerta : mostrarAlerta("No es posible efectuar el pago.")
        activate alerta
        alerta --> frm : mostrar("No es posible efectuar el pago.")
        deactivate alerta
        frm -> Empleado : seleccionarOtroMetodo()
    else Pago exitoso
        venta --> cpv : ventaRegistrada()
        deactivate venta
        cpv -> alerta : mostrarAlerta("Venta concretada correctamente.")
        activate alerta
        alerta --> frm : mostrar("Venta concretada correctamente.")
        deactivate alerta
        frm -> frm : limpiarCamposYActualizar()
    end
end
deactivate cpv

frm -> Empleado : mostrarTicketConfirmacion()
deactivate frm

@enduml
